def test_block_run_when_valid_json_given_and_all_fields_declared() -> None:
    # given
    raw_json = json.dumps({"a": "1", "b": "2"})
    block = JSONParserBlockV1()

    # when
    result = block.run(raw_json=raw_json, expected_fields=["a", "b"])

    # then
    assert result == {
        "error_status": False,
        "a": "1",
        "b": "2",
    }
