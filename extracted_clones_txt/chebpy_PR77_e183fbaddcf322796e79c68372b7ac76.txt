def ufuncTester(ufunc, f, interval, tol):
    a, b = interval
    ff = Chebfun.initfun_adaptive(f, np.linspace(a, b, 13))

    def gg(x):
        return ufunc(f(x))

    GG = getattr(ff, ufunc.__name__)()

    def tester(self):
        xx = interval(self.yy)
        vscl = GG.vscale
        lscl = sum([fun.size for fun in GG])
        self.assertLessEqual(infnorm(gg(xx) - GG(xx)), vscl * lscl * tol)

    return tester
